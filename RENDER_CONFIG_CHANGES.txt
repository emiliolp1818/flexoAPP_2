1. DOCKERFILE (raíz del proyecto) - REEMPLAZAR COMPLETAMENTE
===============================================================================

FROM mcr.microsoft.com/dotnet/sdk:8.0 AS build-env
WORKDIR /app

# Copiar archivos del proyecto
COPY backend/*.csproj ./
RUN dotnet restore

# Copiar todo el código fuente
COPY backend/ ./
RUN dotnet publish -c Release -o out

# Runtime image
FROM mcr.microsoft.com/dotnet/aspnet:8.0
WORKDIR /app
COPY --from=build-env /app/out .

# Crear directorios
RUN mkdir -p logs uploads

# Configuración
ENV ASPNETCORE_ENVIRONMENT=Production
ENV ASPNETCORE_URLS=http://+:10000
EXPOSE 10000

ENTRYPOINT ["dotnet", "FlexoAPP.API.dll"]

5. PROGRAM.CS - VERIFICAR CONFIGURACIÓN DE PUERTO
===============================================================================

En backend/Program.cs, buscar esta sección y asegurar que esté así:

builder.WebHost.ConfigureKestrel(options =>
{
    options.Limits.MaxRequestBodySize = 52428800; // 50MB
    options.Limits.KeepAliveTimeout = TimeSpan.FromMinutes(2);
    
    // Configurar para escuchar en el puerto de Render
    var port = Environment.GetEnvironmentVariable("PORT") ?? "10000";
    options.ListenAnyIP(int.Parse(port));
});

===============================================================================

builder.Services.AddCors(options =>
{
    options.AddPolicy("AllowRenderOrigins", policy =>
    {
        policy.WithOrigins(
            "https://flexoapp-frontend-v2.onrender.com",
            "http://localhost:4200",
            "http://127.0.0.1:4200"
        )
        .AllowAnyMethod()
        .AllowAnyHeader()
        .AllowCredentials();
    });
});
